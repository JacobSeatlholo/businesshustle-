    42→    42→/*  Animated counter hook                                              */
    43→    43→/* ------------------------------------------------------------------ */
    44→    44→function useCounter(end: number, duration = 2000, startOnView = true) {
    45→    45→  const [count, setCount] = useState(0)
    46→    46→  const ref = useRef<HTMLSpanElement>(null)
    47→    47→  const inView = useInView(ref, { once: true, margin: '-50px' })
    48→    48→
    49→    49→  useEffect(() => {
    50→    50→    if (!inView && startOnView) return
    51→    51→    let start = 0
    52→    52→    const step = Math.ceil(end / (duration / 16))
    53→    53→    const timer = setInterval(() => {
    54→    54→      start += step
    55→    55→      if (start >= end) {
    56→    56→        setCount(end)
    57→    57→        clearInterval(timer)
    58→    58→      } else {
    59→    59→        setCount(start)
    60→    60→      }
    61→    61→    }, 16)
    62→    62→    return () => clearInterval(timer)
    63→    63→  }, [inView, end, duration, startOnView])
    64→    64→
    65→    65→  return { count, ref }
    66→    66→}
