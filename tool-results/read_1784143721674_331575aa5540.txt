    48→/* ------------------------------------------------------------------ */
    49→/*  Animated counter hook                                              */
    50→/* ------------------------------------------------------------------ */
    51→function useCounter(end: number, duration = 2000, startOnView = true) {
    52→  const [count, setCount] = useState(0)
    53→  const ref = useRef<HTMLSpanElement>(null)
    54→  const inView = useInView(ref, { once: true, margin: '-50px' })
    55→
    56→  useEffect(() => {
    57→    if (!inView && startOnView) return
    58→    let start = 0
    59→    const step = Math.ceil(end / (duration / 16))
    60→    const timer = setInterval(() => {
    61→      start += step
    62→      if (start >= end) {
    63→        setCount(end)
    64→        clearInterval(timer)
    65→      } else {
    66→        setCount(start)
    67→      }
    68→    }, 16)
    69→    return () => clearInterval(timer)
    70→  }, [inView, end, duration, startOnView])
    71→
    72→  return { count, ref }
    73→}

    91→      initial={{ opacity: 0, y: 30 }}

    93→      transition={{ duration: 0.6 }}
    94→      className={className}
    95→    >
    96→      {children}
    97→    </motion.div>
    98→  )
